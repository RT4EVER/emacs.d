** Copy Vterm
#+begin_src emacs-lisp
  ;; 通过 OSC 52 把复制的内容发给终端剪贴板（需终端或 tmux 支持 OSC 52）。
  ;; 注意：Emacs（包括 31）不存在 `kill-ring-save-hook'，
  ;; 旧版挂在该 hook 上实际从未生效，这里改用 advice 挂在 kill-ring-save 之后。
  (defun my/osc52-copy ()
    "把最近一次 kill 的内容通过 OSC 52 发给终端剪贴板。"
    (let ((text (ignore-errors (substring-no-properties (current-kill 0)))))
      (when (and text (not (display-graphic-p)))
        (ignore-errors
          (send-string-to-terminal
           (format "\e]52;c;%s\a"
                   (base64-encode-string (encode-coding-string text 'utf-8) t)))))))

  (advice-add 'kill-ring-save :after #'my/osc52-copy)
